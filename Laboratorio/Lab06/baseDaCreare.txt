--calendario ora è univoco per insegnamento, prof e data così da avere 
  più di un esame per ogni materia, più esami per ogni professore e 
  più professori per un tipo d'esame 
create table calendario_esami(
	id_ins varchar not null references insegnamenti(id_ins),
	id_prof varchar not null references docenti(id_prof),
	data timestamp not null,
	durata int check(durata between 1 and 6),
	tipo tipo_esame not null,
	primary key(id_ins,id_prof,data,tipo)
);

--funzione per vedere TUTTI gli esami svolti da uno studente
CREATE FUNCTION
esami_svolti(id varchar)
RETURNS refcursor
AS $$
DECLARE
	ref refcursor;	
BEGIN
	open ref for
		select id_ins as "codice esame", nome_corso as "nome esame"
		from esami inner join insegnamenti on esami.id_ins=insegnamenti.id_ins
		where id_stud=$1;
	return ref;
END;
$$ LANGUAGE plpgsql

--funzione che calcola la media di uno studente
CREATE FUNCTION
media_studente(id varchar)
RETURNS float
AS $$
BEGIN
        select avg(voto)
        from esami
        where id_stud=$1 and iscritto=TRUE;
	
END;
$$ LANGUAGE plpgsql

--funzione per vedere se un professore tiene al massimo 3 corsi
CREATE FUNCTION
quanti_corsi(id varchar)
RETURNS varchar
AS $$
DECLARE
        n_corsi integer;
BEGIN
        select count(*) into n_corsi
        from insegnamenti
        where id_prof=$1;

        IF n_corsi>3 THEN
             RETURN 'Il professore tiene troppi corsi';
        ELSE
             RETURN 'Tutto ok';
        END IF;
END;
$$ LANGUAGE plpgsql;
