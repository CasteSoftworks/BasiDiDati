check_person_location
	su inserimento o modifica di una tupla di location, verificare che
	la persona non abbia già una tupla con il medesimo d_role

	verificare che il db sia in stato consistent rispetto alla richiesta
	esiste una persona che abbia diversi country per il medesimo d_role


					*PROCEDURA*
IO FACCIO UNA INSERST O UPDATE DI UN TIZIO
IL TRIGGER SI ATTIVA *PRIMA* DELLA ESECUZIONE DELLA INSERT/UPDATE E LANCIA LA
FUNZIONE DI CONTROLLO DI PRESENZA
	SE IL CONTROLLO è VERO NON FA NULLA (RITORNA NULL)
	ALTRIMENTI RITORNA IL TIZIO
SE è RITONRANTO IL TIZIO ALLORA FACCIO L'INSERT/UPDATE


--consistenza
select person, d_role
from location
group by person, d_role having count(*)>1; 

--funzione di controllo
CREATE OR REPLACE FUNCTION check_person_location() RETURNS TRIGGER AS $$
	DECLARE
	BEGIN
	--controllo se persona e evento sono già presenti
	--SE presenti torno NULL
	--ALTRIMENTI ritorno la tupla da inserire (NEW)

	PERFORM * from location where person=NEW.person AND d_role=NEW.d_role;

	IF FOUND THEN 
		RAISE EXCEPTION 'OPERAZIONE FALLITA: PERSONA GIà PRESENTE NEL db PER LO EVENTO INDICATO';
		RETURN NULL;
	ELSE
		RETURN NEW;
	END IF;

	END;
$$ language plpgsql;

--trigger
create trigger person_location_trigger before insert or update on location
for each row execute procedure check_person_location();



--si crei un trigger che ricalcoli il valore di scoreal cambio di scala di un
rating
CREATE OR REPLACE FUNCTION scale_update() RETURNS TRIGGER AS $$
	DECLARE
		new_score rating.score%TYPE;
        BEGIN
		IF (OLD.scale <> NEW.scale) THEN
			new_score:=(OLD.score / OLD.scale) * NEW.scale;
			NEW.score:=new_score;
		END IF;
		RETURN NEW;
	END;
$$ language plpgsql;

CREATE TRIGGER update_scale_rating BEFORE UPDATE ON rating FOR EACH ROW EXECUTE
PROCEDURE scale_update();


--check_character
--su aggiunta o inserimento di tuple in crew
--il trigger elimina eventuali nomi di personaggi per quelle categorie (p_role)
--che non sono actor

CREATE OR REPLACE FUNCTION check_character() RETURNS TRIGGER AS $$
        DECLARE
        BEGIN
		IF NEW.p_role <>'actor' THEN
			NEW.character:='';
		END IF;
		RETURN NEW;
        END;
$$ language plpgsql;

CREATE TRIGGER character_check_trigger BEFORE INSERT OR UPDATE ON crew FOR EACH ROW
EXECUTE PROCEDURE check_character();


--person_stats
-- sic costruisca una vista materializzata per ogni persona il numero di
pellicole nelle quali ha lavorato per ogni ruolo
--la vista prevede person.id, person.given name, person.p_role, performance
number

CREATE TABLE person_stats(
	person_id varchar(10) references person(id) on update cascade on delete cascade,
	given_name varchar(100) NOT null,
	person_role person_role,
	perf_number integer,
	primary key(person_id,person_role)
);

--funzione di setup iniziale
--la riempio a casa
--uso group by e count

CREATE OR REPLACE FUNCTION p_stats_setup() RETURNS SETOF person_stats AS $$
DECLARE
	a_per person_stats%rowtype;
BEGIN
	for a_per in select id,given_name,p_role,count(*) from person inner join crew on crew.person = person.id group by id,given_name,p_role order by 1
	loop
		return next a_per;
	end loop;
END;
$$ language plpgsql;

insert into person_stats
select * from p_stats_setup();

--funzione di update
--O reinizio da 0 ogni volta e riconteggio/cancello/inserisco da nuovo
--O controllo se ci sono o solo NEW, o solo OLD o ENTRAMBI
CREATE OR REPLACE FUNCTION update_person_stats() RETURNS TRIGGER AS $$
	DECLARE
		p_id crew.person%TYPE;
		p_ro crew.p_role%TYPE;
	BEGIN
		IF (TG_OP='DELETE') NULL THEN
			--è una eliminazione
			p_id:=OLD.person;
			p_ro:=OLD.p_role;
		ELSE
			--è un aggiornamneto/inserimento
			p_id:=NEW.person;
                        p_ro:=NEW.p_role;
		END IF;

		num_perf:=0;
		for a_crew in select * FROM crew where person=p_id AND p_role=p_ro
		LOOP
			num_perf:=num_perf+1;
		END LOOP;

		--verifico che esista o meno in person_stats
		perform * from person_stats where person_id=p_id and
		person_role=p_ro

		IF FOUND THEN
			UPDATE person_stats SET perf_number=num_perf where
			person_id=p_id AND person_role=p_ro
		ELSE
			select given_name into g_name from person were
			id=person_id

			insert into
			person_stats(person_id,given_name,person_role,performance_number)
			VALUES (p_id,g_name,p_ro,num_perf);
		END IF;

		RETURN NULL; --è una after, non necessita di altro
	END;

$$ language plpgsql;


--trigger AFTER
--DOPO l'inserimento in crew DEVO aggiornare stats
CREATE TRIGGER update_person_stat_trigger AFTER INSERT OR UPDATE OR DELETE ON
crew FOR EACH ROW EXECUTE PROCEDURE update_person_stats();
